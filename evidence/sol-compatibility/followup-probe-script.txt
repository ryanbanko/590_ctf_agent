import argparse
import json
import os
from pathlib import Path
import time
import uuid

import httpx2
from langfuse import Langfuse
from langfuse.openai import OpenAI

parser = argparse.ArgumentParser()
parser.add_argument('--variant', choices=['minimal', 'runner', 'tool', 'production', 'recorded-first-request'], required=True)
parser.add_argument('--model', default='gpt-6.1-sol')
args = parser.parse_args()
request = {'model': args.model, 'input': 'Reply with exactly OK.',
           'max_output_tokens': 128, 'store': False}
if args.variant != 'minimal':
    request.update(reasoning={'effort': 'medium', 'summary': 'auto'},
                   include=['reasoning.encrypted_content'], parallel_tool_calls=False)
if args.variant == 'tool':
    request['input'] = 'Call diagnostic_echo with message OK. This is a synthetic API compatibility check.'
    request['tools'] = [{'type': 'function', 'name': 'diagnostic_echo',
                         'description': 'Return a synthetic message; no tool will be executed.',
                         'parameters': {'type': 'object', 'properties': {'message': {'type': 'string'}},
                                        'required': ['message'], 'additionalProperties': False}, 'strict': False}]

if args.variant == 'production':
    import sys
    from copy import deepcopy
    sys.path.insert(0, str(Path.cwd()))
    from ctf_agent.agent import SYSTEM_PROMPT
    from ctf_agent.tools import TOOL_DEFINITIONS
    request['input'] = [{'role': 'system', 'content': SYSTEM_PROMPT},
                        {'role': 'user', 'content': 'This is a synthetic API compatibility check, not a challenge. Return exactly OK without calling tools.'}]
    request['max_output_tokens'] = 8192
    request['tools'] = [{'type': 'function', **deepcopy(tool), 'strict': False} for tool in TOOL_DEFINITIONS]
    next(tool for tool in request['tools'] if tool['name'] == 'run_command')['parameters']['properties']['timeout_seconds'].update(maximum=120.0, default=120.0)

if args.variant == 'recorded-first-request':
    import sys
    sys.path.insert(0, str(Path.cwd()))
    from ctf_agent.config import load_experiment
    _, settings = load_experiment(Path('configs/htb-sol.toml'))
    rows = json.loads(Path('runs/htb/attempts/challenge-14d10d57219e/observations.json').read_text())['data']
    generation = next(row for row in rows if row['type'] == 'GENERATION')
    native = generation['input']
    if isinstance(native, str):
        native = json.loads(native)
    request = {'model': settings['model'], 'input': native['input'], 'tools': native['tools'],
               'parallel_tool_calls': native['parallel_tool_calls'],
               'reasoning': {'effort': settings['reasoning_effort'], 'summary': 'auto'},
               'max_output_tokens': settings['max_output_tokens'], 'store': False,
               'include': ['reasoning.encrypted_content']}

secrets = [os.environ[k] for k in ('LITELLM_TOKEN', 'LANGFUSE_SECRET_KEY') if os.getenv(k)]
record = {'variant': args.variant, 'request': request, 'model_calls': 0,
          'tools_executed': 0, 'usage': None, 'http_error': None}

def safe(value):
    text = str(value)
    for secret in secrets:
        text = text.replace(secret, '[REDACTED]')
    return text[:2000]

def error_hook(response):
    if response.is_error:
        response.read()
        try:
            data = response.json()
            error = data.get('error', {}) if isinstance(data, dict) else {}
            if not isinstance(error, dict):
                error = {'message': error}
        except ValueError:
            error = {'message': 'Non-JSON error body withheld'}
        record['http_error'] = {'http_status': response.status_code,
                               **{k: safe(error[k]) for k in ('message', 'type', 'code', 'param') if k in error}}
        # Native tracing receives only the status, never the raw provider body.
        raise RuntimeError(f'gateway_http_{response.status_code}')

directory = Path('runs/sol-compatibility') / ('probe-' + uuid.uuid4().hex[:12])
directory.mkdir(parents=True, mode=0o700)
started = time.monotonic()
lf = Langfuse(timeout=10)
try:
    if not lf.auth_check():
        raise RuntimeError('Langfuse authentication failed before model request')
    with lf.start_as_current_observation(name='sol-compatibility', input=request,
            metadata={'variant': args.variant, 'synthetic': True}) as observation:
        record['trace_id'] = observation.trace_id
        try:
            with OpenAI(api_key=os.environ['LITELLM_TOKEN'], base_url='https://litellm.oit.duke.edu/v1',
                        max_retries=0, timeout=20,
                        http_client=httpx2.Client(event_hooks={'response': [error_hook]})) as client:
                record['model_calls'] = 1
                response = client.responses.create(**request, name='model-response',
                    langfuse_public_key=os.environ['LANGFUSE_PUBLIC_KEY'])
                record.update(status=response.status, output_text=safe(response.output_text),
                    output_types=[item.type for item in response.output],
                    usage=response.usage.model_dump(mode='json') if response.usage else None)
        except Exception as error:
            record['exception'] = type(error).__name__
        observation.update(output=record)
    record['trace_url'] = lf.get_trace_url(trace_id=record['trace_id'])
finally:
    lf.shutdown()
    record['seconds'] = round(time.monotonic() - started, 6)
    (directory / 'probe.json').write_text(json.dumps(record, indent=2) + '\n')
    print(json.dumps({'record': str(directory / 'probe.json'), **record}, indent=2))
